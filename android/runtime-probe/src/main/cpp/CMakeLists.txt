cmake_minimum_required(VERSION 3.22.1)
project(st_runtime_probe LANGUAGES CXX)
if(NOT ANDROID_ABI STREQUAL "arm64-v8a")
    message(FATAL_ERROR "Only arm64-v8a is supported")
endif()
if(NOT EXISTS "${NODE_ARTIFACT_DIR}/lib/libnode.so")
    message(FATAL_ERROR "Build Node first: python3 tools/build_node.py")
endif()
add_library(node SHARED IMPORTED)
set_target_properties(node PROPERTIES IMPORTED_LOCATION "${NODE_ARTIFACT_DIR}/lib/libnode.so")
add_library(stprobe SHARED node_bridge.cpp)
target_compile_features(stprobe PRIVATE cxx_std_20)
target_include_directories(stprobe SYSTEM PRIVATE "${NODE_ARTIFACT_DIR}/include/node")
target_compile_options(stprobe PRIVATE -Wall -Wextra -Werror)
target_link_options(stprobe PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--no-undefined")
target_link_libraries(stprobe PRIVATE node log)
